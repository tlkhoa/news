-- FUNCTION: public.get_rate_success_bu_app_and_minutes(text, integer)

-- DROP FUNCTION IF EXISTS public.get_rate_success_bu_app_and_minutes(text, integer);

CREATE OR REPLACE FUNCTION public.get_rate_success_bu_app_and_minutes(
	p_appid text,
	p_minutes integer)
    RETURNS double precision
    LANGUAGE 'plpgsql'
    COST 100
    VOLATILE PARALLEL UNSAFE
AS $BODY$
DECLARE
    success_count INT;
	fail_count INT;
	success_rate DOUBLE PRECISION;
BEGIN
   IF p_appId= 'com.appmusicdownloader.fdownloadmusicfree2' THEN
      SELECT COUNT(*)
        INTO success_count
        FROM events_com_appmusicdownloader_fdownloadmusicfree2
       WHERE event_name = 'gls'
         AND event_time >= NOW() - (p_minutes || ' minutes')::INTERVAL;
      SELECT COUNT(*)
        INTO fail_count
        FROM events_com_appmusicdownloader_fdownloadmusicfree2
       WHERE event_name = 'glf'
         AND event_time >= NOW() - (p_minutes || ' minutes')::INTERVAL;		 
   END IF;

   IF (success_count + fail_count) > 0 THEN
      success_rate := ((success_count::DOUBLE PRECISION / (success_count + fail_count)) * 100)::DOUBLE PRECISION;
   ELSE
      success_rate := 0; -- Tránh chia cho 0
   END IF;

   RETURN 99.9;-- success_rate::DOUBLE PRECISION;
END;
$BODY$;

ALTER FUNCTION public.get_rate_success_bu_app_and_minutes(text, integer)
    OWNER TO postgres;
